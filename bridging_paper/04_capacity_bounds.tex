\section{Capacity Bounds}
\label{sec:capacity}

We establish tight bounds on the number of objects encodable in superposition as a function of dimension and sparsity. Our analysis builds on the Welch bound from coding theory.

\subsection{Preliminaries}

We recall key definitions from Section~\ref{sec:equivalence} and introduce additional notation.

\begin{definition}[Coherence]
\label{def:coherence}
For encoding $\mathbf{W} \in \R^{d \times n}$ with unit-norm columns, the \emph{coherence} is:
\begin{equation}
\mu(\mathbf{W}) := \max_{i \neq j} |\langle \mathbf{w}_i, \mathbf{w}_j \rangle|
\end{equation}
\end{definition}

\begin{definition}[Effective Dimension]
\label{def:eff_dim}
For encoding $\mathbf{W}$ and threshold $\theta > 0$, the \emph{effective number of encoded objects} is:
\begin{equation}
n_{\text{eff}}(\mathbf{W}, \theta) := |\{i \in [n] : \|\mathbf{w}_i\| \geq \theta\}|
\end{equation}
We write $n_{\text{eff}}$ when $\theta = 1/2$ is implicit.
\end{definition}

\begin{definition}[Interference Budget]
\label{def:budget}
For co-occurrence matrix $P$ and encoding $\mathbf{W}$, the \emph{interference budget} is:
\begin{equation}
\mathcal{B}(\mathbf{W}, P) := \sum_{i \neq j} P_{ij} \langle \mathbf{w}_i, \mathbf{w}_j \rangle^2
\end{equation}
\end{definition}

\subsection{The Welch Bound}

The following classical result from coding theory constrains how many nearly-orthogonal vectors can exist in a given dimension.

\begin{theorem}[Welch Bound \citep{welch1974lower}]
\label{thm:welch}
For any collection of $n$ unit vectors $\mathbf{w}_1, \ldots, \mathbf{w}_n \in \R^d$ with $n > d$:
\begin{equation}
\sum_{i \neq j} \langle \mathbf{w}_i, \mathbf{w}_j \rangle^2 \geq \frac{n^2 - nd}{d}
\label{eq:welch_sum}
\end{equation}
Equivalently, the maximum coherence satisfies:
\begin{equation}
\mu(\mathbf{W}) \geq \sqrt{\frac{n - d}{d(n-1)}}
\label{eq:welch_max}
\end{equation}
Equality holds if and only if $\mathbf{W}$ forms an \emph{equiangular tight frame}.
\end{theorem}

\begin{proof}
Let $\mathbf{G} = \mathbf{W}^\top \mathbf{W}$ be the Gram matrix. Since $\|\mathbf{w}_i\| = 1$, we have $G_{ii} = 1$ for all $i$.

The trace satisfies:
\begin{equation}
\text{tr}(\mathbf{G}) = n, \quad \text{tr}(\mathbf{G}^2) = \|\mathbf{G}\|_F^2 = \sum_{i,j} G_{ij}^2
\end{equation}

Since $\mathbf{G} = \mathbf{W}^\top \mathbf{W}$ has rank at most $d$, let $\lambda_1, \ldots, \lambda_d$ be its nonzero eigenvalues. Then:
\begin{align}
\text{tr}(\mathbf{G}) &= \sum_{k=1}^d \lambda_k = n \\
\text{tr}(\mathbf{G}^2) &= \sum_{k=1}^d \lambda_k^2 \geq \frac{1}{d}\left(\sum_{k=1}^d \lambda_k\right)^2 = \frac{n^2}{d}
\end{align}
where the inequality is Cauchy-Schwarz.

Expanding $\text{tr}(\mathbf{G}^2)$:
\begin{equation}
\sum_{i,j} G_{ij}^2 = \sum_i G_{ii}^2 + \sum_{i \neq j} G_{ij}^2 = n + \sum_{i \neq j} G_{ij}^2 \geq \frac{n^2}{d}
\end{equation}

Rearranging: $\sum_{i \neq j} G_{ij}^2 \geq \frac{n^2}{d} - n = \frac{n^2 - nd}{d} = \frac{n(n-d)}{d}$.

For the maximum coherence, note that:
\begin{equation}
\sum_{i \neq j} G_{ij}^2 \leq n(n-1) \mu^2
\end{equation}
so $\mu^2 \geq \frac{n(n-d)}{d \cdot n(n-1)} = \frac{n-d}{d(n-1)}$.
\end{proof}

\subsection{Main Capacity Result}

\begin{theorem}[Capacity-Sparsity Upper Bound]
\label{thm:capacity}
Let $\mathbf{W}^* = \arg\min_{\mathbf{W}} \mathcal{L}(\mathbf{W}, P)$ subject to $\mathcal{B}(\mathbf{W}, P) \leq \tau$ for interference tolerance $\tau > 0$. Then:
\begin{equation}
n_{\text{eff}}(\mathbf{W}^*) \leq d + \sqrt{\frac{d(d-1) \cdot \tau}{1 - S(P)}}
\label{eq:capacity_upper}
\end{equation}
For sparse co-occurrence ($S(P) \to 1$), this simplifies to:
\begin{equation}
n_{\text{eff}} = O\left(d + \frac{d}{\sqrt{(1-S)\tau}}\right) = O\left(\frac{d^2 \cdot S}{\tau}\right)
\end{equation}
\end{theorem}

\begin{proof}
\textbf{Step 1: Normalization.}
Without loss of generality, consider unit-norm encodings (rescaling affects $\mathcal{L}_{\text{cap}}$ but not the combinatorial structure). Let $n' = n_{\text{eff}}$ be the number of effectively encoded objects.

\textbf{Step 2: Applying Welch bound.}
By Theorem~\ref{thm:welch}, the total squared inner product satisfies:
\begin{equation}
\sum_{i \neq j} G_{ij}^2 \geq \frac{n'(n' - d)}{d}
\end{equation}

\textbf{Step 3: Relating to interference budget.}
The interference budget is:
\begin{equation}
\mathcal{B}(\mathbf{W}, P) = \sum_{i \neq j} P_{ij} G_{ij}^2
\end{equation}

Let $\bar{P} := \frac{1}{n'(n'-1)} \sum_{i \neq j} P_{ij}$ be the average co-occurrence among effective objects. By definition of sparsity, $\bar{P} = 1 - S(P)$.

By convexity (or directly):
\begin{equation}
\mathcal{B}(\mathbf{W}, P) \geq \bar{P} \cdot \min_{\text{unit } \mathbf{W}} \sum_{i \neq j} G_{ij}^2 = (1-S) \cdot \frac{n'(n'-d)}{d}
\end{equation}

\textbf{Step 4: Enforcing the constraint.}
The constraint $\mathcal{B}(\mathbf{W}, P) \leq \tau$ requires:
\begin{equation}
(1-S) \cdot \frac{n'(n'-d)}{d} \leq \tau
\end{equation}

Rearranging:
\begin{equation}
n'(n' - d) \leq \frac{d\tau}{1-S}
\end{equation}

This is a quadratic in $n'$. Solving:
\begin{equation}
n' \leq \frac{d + \sqrt{d^2 + \frac{4d\tau}{1-S}}}{2} \leq d + \sqrt{\frac{d\tau}{1-S}}
\end{equation}

For the asymptotic form, when $S \to 1$:
\begin{equation}
n' \leq d + \sqrt{\frac{d\tau}{1-S}} \approx d + d\sqrt{\frac{\tau}{1-S}}
\end{equation}

Setting $\tau = O(1)$ and $1-S = O(p^2)$ for activation probability $p$:
\begin{equation}
n' = O(d/p) = O(d^2 S / \tau)
\end{equation}
using $S = 1 - p^2 \approx 1 - 1/d$ for the critical regime.
\end{proof}

\begin{theorem}[Capacity-Sparsity Lower Bound]
\label{thm:capacity_lower}
For any sparsity $S \in (0,1)$ and tolerance $\tau > 0$, there exists an encoding $\mathbf{W} \in \R^{d \times n}$ achieving:
\begin{equation}
n_{\text{eff}}(\mathbf{W}) \geq c \cdot d \cdot \sqrt{\frac{S}{\tau(1-S)}}
\label{eq:capacity_lower}
\end{equation}
for a universal constant $c > 0$.
\end{theorem}

\begin{proof}
\textbf{Construction using equiangular tight frames.}

When they exist, equiangular tight frames (ETFs) achieve the Welch bound with equality. For $n$ vectors in $\R^d$ with $n \leq d(d+1)/2$, an ETF has:
\begin{equation}
|\langle \mathbf{w}_i, \mathbf{w}_j \rangle| = \sqrt{\frac{n-d}{d(n-1)}} =: \mu^* \quad \text{for all } i \neq j
\end{equation}

\textbf{Interference calculation.}
For an ETF with co-occurrence matrix $P$ having average off-diagonal value $\bar{P} = 1 - S$:
\begin{align}
\mathcal{B}(\mathbf{W}, P) &= \sum_{i \neq j} P_{ij} (\mu^*)^2 \\
&\leq (1-S) \cdot n(n-1) \cdot \frac{n-d}{d(n-1)} \\
&= (1-S) \cdot \frac{n(n-d)}{d}
\end{align}

Setting $\mathcal{B} \leq \tau$ and solving for $n$:
\begin{equation}
n \geq d + \Omega\left(\sqrt{\frac{d\tau}{1-S}}\right)
\end{equation}

\textbf{Explicit construction for $d = 2$.}
Place $n = 2k$ vectors as vertices of a regular $k$-gon and its reflection:
\begin{equation}
\mathbf{w}_j = (\cos(2\pi j/k), \sin(2\pi j/k))^\top, \quad j = 1, \ldots, k
\end{equation}
and $\mathbf{w}_{k+j} = (\cos(2\pi j/k), -\sin(2\pi j/k))^\top$.

The inner products are $\langle \mathbf{w}_i, \mathbf{w}_j \rangle = \cos(2\pi(i-j)/k)$ for vectors in the same half, and these average to $O(1/k)$ for large $k$.

For sparse $P$ with $P_{ij} = p^2$ (independent activation):
\begin{equation}
\mathcal{B} = p^2 \sum_{i \neq j} \cos^2(2\pi(i-j)/k) = O(p^2 k)
\end{equation}

Setting $\mathcal{B} = \tau$: $k = O(\tau/p^2)$, so $n = 2k = O(\tau/p^2) = O(d^2 S/\tau)$ for $d = 2$.

The construction generalizes to higher dimensions using products of regular polygons or random constructions that approach the Welch bound.
\end{proof}

\begin{corollary}[Tight Characterization]
\label{cor:tight}
The capacity scales as:
\begin{equation}
n_{\text{eff}} = \Theta\left(d + \frac{d}{\sqrt{(1-S)\tau}}\right)
\end{equation}
with matching upper (Theorem~\ref{thm:capacity}) and lower (Theorem~\ref{thm:capacity_lower}) bounds up to constants.
\end{corollary}

\subsection{Implications}

\begin{corollary}[Critical Sparsity Threshold]
\label{cor:critical}
Superposition ($n_{\text{eff}} > d$) is beneficial if and only if:
\begin{equation}
S(P) > 1 - \frac{d\tau}{(n-d)^2}
\end{equation}
Below this threshold, orthogonal encoding is optimal.
\end{corollary}

\begin{proof}
Superposition provides benefit when $n_{\text{eff}} > d$. From Theorem~\ref{thm:capacity}, this requires:
\begin{equation}
\sqrt{\frac{d\tau}{1-S}} > 0 \implies \frac{d\tau}{1-S} > (n_{\text{eff}} - d)^2
\end{equation}
Rearranging gives the stated condition.
\end{proof}

\begin{corollary}[Scaling Laws]
\label{cor:scaling}
For the two superposition settings:
\begin{enumerate}
    \item[(a)] \textbf{Representational:} With i.i.d.\ activation probability $p$, sparsity is $S = 1 - p^2$, giving:
    \begin{equation}
    n_{\text{eff}} = O\left(\frac{d}{p}\right) = O\left(\frac{d}{\sqrt{1-S}}\right)
    \end{equation}

    \item[(b)] \textbf{Reasoning:} For a branching problem with $K$ total traces and co-validity probability $q = \Pr[\text{both valid}]$, sparsity is $S = 1 - q$, giving:
    \begin{equation}
    n_{\text{eff}} = O\left(\frac{d}{\sqrt{q}}\right)
    \end{equation}
\end{enumerate}
\end{corollary}

\begin{remark}[Comparison to Prior Work]
\label{rem:prior}
Our bounds recover and extend prior results:
\begin{itemize}
    \item \citet{elhage2022toy} empirically observed $n \propto 1/p$ scaling; Theorem~\ref{thm:capacity} provides a rigorous derivation.
    \item The Welch bound connection explains why the $O(d^2)$ scaling is fundamental: it arises from the geometry of nearly-orthogonal vectors, not the specific loss function.
    \item The role of sparsity $S$ is made precise: it multiplicatively amplifies the geometric capacity.
\end{itemize}
\end{remark}

\subsection{Local vs.\ Global Sparsity}

\begin{remark}[Frontier Growth in Reasoning]
\label{rem:frontier}
In reasoning superposition, the ``frontier'' of active traces at step $t$ may have size $|F_t| = O(b^t)$ for branching factor $b$, potentially exceeding the capacity bound.

This is resolved by distinguishing:
\begin{enumerate}
    \item \textbf{Global sparsity $S$}: Over all trace pairs, most never both lead to valid solutions.
    \item \textbf{Local budget $\mathcal{B}^{(t)}$}: At step $t$, interference among frontier traces.
\end{enumerate}

The model succeeds if either:
\begin{itemize}
    \item $|F_t| \leq n_{\text{eff}}(d, S^{(t)}, \tau)$ for all $t$ (frontier fits in capacity), or
    \item Iterative refinement (Section~\ref{sec:timespace}) prunes the frontier before overflow.
\end{itemize}
\end{remark}
